const { app, BrowserWindow, ipcMain, dialog, shell } = require("electron");
const path = require("path");
const Store = require("electron-store");

const fs = require("fs");

const store = new Store();

let hostWindow = null;
let playerWindow = null;
let musicWatcher = null;

// --- Criar janela HOST ----------------------------------------
function createHostWindow() {
  hostWindow = new BrowserWindow({
    width: 1280,
    height: 800,
    minWidth: 1024,
    minHeight: 700,
    title: "Voxly - Gerencia",
    webPreferences: {
      nodeIntegration: false,
      contextIsolation: true,
      preload: path.join(__dirname, "preload.js"),
    },
  });
  hostWindow.loadFile(path.join(__dirname, "screens", "host.html"));
  hostWindow.on("closed", () => {
    hostWindow = null;
    app.quit();
  });
}

// --- Criar janela PLAYER --------------------------------------
function createPlayerWindow() {
  playerWindow = new BrowserWindow({
    width: 1920,
    height: 1080,
    title: "Voxly - Palco",
    fullscreenable: true,
    frame: false,
    backgroundColor: "#000000",
    webPreferences: {

      contextIsolation: true,
      preload: path.join(__dirname, "preload.js"),
    },
  });
  playerWindow.loadFile(path.join(__dirname, "screens", "player.html"));

app.whenReady().then(() => {
  createHostWindow();
  createPlayerWindow();

app.on("window-all-closed", () => {
  if (process.platform !== "darwin") app.quit();
});

// --- IPC: Selecionar pasta de musicas -------------------------
ipcMain.handle("select-music-folder", async () => {
  const result = await dialog.showOpenDialog(hostWindow, {
    properties: ["openDirectory"],
    title: "Selecionar pasta de musicas",
  });
  if (!result.canceled && result.filePaths.length > 0) {
    const folder = result.filePaths[0];
    store.set("musicFolder", folder);
    startWatcher(folder);
    return folder;
  }
  return null;
});

ipcMain.handle("get-music-folder", () => store.get("musicFolder", null));

// --- IPC: Listar musicas na pasta -----------------------------
ipcMain.handle("list-music-files", () => {
  const folder = store.get("musicFolder", null);
  if (!folder || !fs.existsSync(folder)) return [];
  const exts = [".mp4", ".mkv", ".avi", ".webm", ".mp3"];
  return fs.readdirSync(folder)
    .filter(f => exts.includes(path.extname(f).toLowerCase()))
    .map(f => ({ name: f, fullPath: path.join(folder, f) }));
});

// --- IPC: Resolver arquivo de uma musica ----------------------
ipcMain.handle("resolve-music-file", (_, songName, artist) => {
  const folder = store.get("musicFolder", null);
  if (!folder) return null;
  const exts = [".mp4", ".mkv", ".avi", ".webm"];
  const files = fs.readdirSync(folder);
  const query = `${songName} ${artist}`.toLowerCase().replace(/\s+/g, "");
  for (const f of files) {
    const base = path.basename(f, path.extname(f)).toLowerCase().replace(/\s+/g, "");
    if (base.includes(query) || query.includes(base)) {
      return path.join(folder, f);
    }
  }
  return null;
});

// --- IPC: Vincular arquivo manualmente a uma musica -----------
ipcMain.handle("link-music-file", async (_, songId) => {
  const result = await dialog.showOpenDialog(hostWindow, {
    properties: ["openFile"],
    title: "Selecionar arquivo da musica",
    filters: [
      { name: "Videos", extensions: ["mp4", "mkv", "avi", "webm"] },
      { name: "Todos os arquivos", extensions: ["*"] },
    ],
  });
  if (!result.canceled && result.filePaths.length > 0) {
    const links = store.get("musicLinks", {});
    links[songId] = result.filePaths[0];
    store.set("musicLinks", links);
    return result.filePaths[0];
  }
  return null;
});

ipcMain.handle("get-music-links", () => store.get("musicLinks", {}));

// --- IPC: Reproduzir musica na janela player ------------------
ipcMain.handle("play-song", (_, filePath) => {
  if (playerWindow) {
    playerWindow.webContents.send("play-video", filePath);
  }
});

ipcMain.handle("player-command", (_, cmd) => {
  if (playerWindow) playerWindow.webContents.send("player-cmd", cmd);
});

// --- IPC: QR Code config -------------------------------------
ipcMain.handle("get-webapp-url", () => store.get("webAppUrl", ""));
ipcMain.handle("set-webapp-url", (_, url) => store.set("webAppUrl", url));

// --- Watcher de pasta de musicas ------------------------------
function startWatcher(folder) {
  if (musicWatcher) musicWatcher.close();
  musicWatcher = chokidar.watch(folder, { ignoreInitial: true });
  musicWatcher.on("add", () => {
    if (hostWindow) hostWindow.webContents.send("music-folder-changed");
  });
  musicWatcher.on("unlink", () => {
    if (hostWindow) hostWindow.webContents.send("music-folder-changed");
  });
}

// Retomar watch ao iniciar se pasta ja estava configurada
app.whenReady().then(() => {
  const folder = store.get("musicFolder");
  if (folder && fs.existsSync(folder)) startWatcher(folder);
});

async function refreshQueueFiles() {
  const links = await window.electronAPI.getMusicLinks();
  for (const item of queueData) {
    if (links[item.id]) {
      item._resolvedFile = links[item.id];
      item._missing = false;
    } else {
      const fp = await window.electronAPI.resolveMusicFile(item.song, item.artist);
      item._resolvedFile = fp;
      item._missing = !fp;
    }
  }
  renderQueue();
}

// -- Render fila --------------------------------------------------------------
function renderQueue() {
  const el = document.getElementById("queue-list");
  if (!queueData.length) { el.innerHTML = '<div class="empty">A fila esta vazia.<br>Aguarde solicitacoes.</div>'; return; }
  el.innerHTML = queueData.map((item, i) => {
    const isPlaying = currentSong?.id === item.id;
    const isMissing = item._missing;
    const isDuet    = item.duetWith?.length > 0;
    let cls = "queue-item";
    if (isPlaying) cls += " playing";
    if (isMissing) cls += " missing";
    if (isDuet)    cls += " duet";

    let badge = "";
    if (isPlaying) badge = '<span class="queue-badge playing">TOCANDO</span>';
    else if (isMissing) badge = '<span class="queue-badge">SEM ARQUIVO</span>';
    else if (isDuet)    badge = '<span class="queue-badge duet">DUETO</span>';

    const duetNames = isDuet ? `<div style="font-size:.72rem;color:var(--warn);margin-top:2px">? Dueto: ${item.duetWith.join(", ")}</div>` : "";

    const actions = isMissing
      ? `<button class="btn btn-warn btn-sm" onclick="linkFile('${item.id}')">? Vincular Arquivo</button>`
      : "";

    const playBtn = (!isPlaying && !isMissing)
      ? `<button class="btn btn-ok btn-sm" onclick="playSong('${item.id}')">> Tocar</button>` : "";

    const removeBtn = `<button class="btn btn-danger btn-sm" onclick="removeSong('${item.id}')">? Remover</button>`;

    return `<div class="${cls}">
      <div class="queue-pos">${i + 1}</div>
      <div class="queue-info">
        <div class="queue-song">${escHtml(item.song)}</div>
        <div class="queue-artist">${escHtml(item.artist)}</div>
        <div class="queue-singer">? ${escHtml(item.singerName || "-")}</div>
        ${duetNames}
        <div class="queue-actions">${playBtn}${actions}${removeBtn}</div>
      </div>
      ${badge}
    </div>`;
  }).join("");
}

    showScreen("auth");
  }
});

function showScreen(name) {
  ["auth-screen","profile-screen","app-screen"].forEach(id => {
    document.getElementById(id).style.display = id.startsWith(name) ? (name === "app" ? "block" : "flex") : "none";
  });
  if (name === "app") document.getElementById("app-screen").style.display = "block";
}

function showApp() {
  showScreen("app");
  document.getElementById("app-name").textContent  = userData.displayName || "-";
  document.getElementById("app-email").textContent = userData.email || currentUser.email;
  if (userData.photoURL) document.getElementById("app-avatar").src = userData.photoURL;
  subscribeQueue();
  subscribeUsers();
  subscribeMyStatus();
}

// -- Fila em tempo real --------------------------------------------------------
function subscribeQueue() {
  onSnapshot(query(collection(db, "queue"), orderBy("requestedAt","asc")), snap => {
    queueData = snap.docs.map(d => ({ id: d.id, ...d.data() }));
    renderMiniQueue();
    updateStatus();
  });
}

function renderMiniQueue() {
  const el = document.getElementById("queue-mini-list");
  if (!queueData.length) { el.innerHTML = '<div style="color:var(--muted);font-size:.85rem">Fila vazia</div>'; return; }
  el.innerHTML = queueData.map((item, i) => `
    <div class="queue-mini-item">
      <span class="pos-badge">${i+1}</span>
      <div style="flex:1;margin-left:10px">
        <div style="font-weight:600">${escHtml(item.song)}</div>
        <div style="font-size:.75rem;color:var(--muted)">${escHtml(item.artist)} . ${escHtml(item.singerName||"")}</div>
      </div>
      ${item.duetWith?.length ? '<span style="font-size:.7rem;color:var(--warn)">DUETO</span>' : ""}
    </div>`).join("");
}

// -- Meu status ----------------------------------------------------------------
function subscribeMyStatus() {
  if (!currentUser) return;
  onSnapshot(doc(db, "users", currentUser.uid), snap => {
    if (!snap.exists()) return;
    userData = snap.data();
    updateStatus();
  });
}

function updateStatus() {
  if (!userData || !currentUser) return;
  const songsUsed  = userData.songsThisCycle  || 0;
  const duetsUsed  = userData.duetsThisCycle  || 0;
  const canSing    = songsUsed  < rules.songsPerCycle;
  const canDuet    = duetsUsed  < rules.duetsPerCycle;
  const myInQueue  = queueData.some(q => q.userId === currentUser.uid);

  const box = document.getElementById("user-status-box");
  const txt = document.getElementById("status-text");
  const req = document.getElementById("request-section");
  const dut = document.getElementById("duet-section");

  if (myInQueue) {
    const pos = queueData.findIndex(q => q.userId === currentUser.uid) + 1;
    box.className = "status-box waiting";
    txt.innerHTML = `? Voce esta na fila na posicao <strong>${pos}</strong>.<br><span style="font-size:.78rem;color:var(--muted)">Aguarde ser chamado!</span>`;
    req.style.display = "none";
    dut.style.display = canDuet ? "block" : "none";
  } else if (!canSing) {
    box.className = "status-box blocked";
    txt.innerHTML = `? Voce ja cantou neste ciclo.<br><span style="font-size:.78rem;color:var(--muted)">${canDuet ? "Voce ainda pode participar de um dueto." : "Aguarde o proximo ciclo."}</span>`;
    req.style.display = "none";
    dut.style.display = canDuet ? "block" : "none";
  } else {
    box.className = "status-box playing";
    txt.innerHTML = `? Voce pode solicitar uma musica!<br><span style="font-size:.78rem;color:var(--muted)">Musicas: ${songsUsed}/${rules.songsPerCycle} . Duetos: ${duetsUsed}/${rules.duetsPerCycle}</span>`;
    req.style.display = "block";
    dut.style.display = canDuet ? "block" : "none";
  }
}

// -- Cantores online (para dueto) ----------------------------------------------
function subscribeUsers() {
  onSnapshot(collection(db, "users"), snap => {
    const others = snap.docs
      .map(d => ({ id: d.id, ...d.data() }))
      .filter(u => u.id !== currentUser?.uid);
    renderDuetList(others);
  });
}

function renderDuetList(users) {
  const el = document.getElementById("duet-list");
  if (!users.length) { el.innerHTML = '<div style="color:var(--muted);font-size:.85rem">Nenhum outro cantor presente.</div>'; return; }
  el.innerHTML = users.map(u => `
    <div class="duet-item ${selectedDuetIds.includes(u.id)?"selected":""}" onclick="toggleDuet('${u.id}', this)">
      <img class="duet-avatar" src="${u.photoURL||"data:image/svg+xml,<svg xmlns='http://www.w3.org/2000/svg'><rect width='36' height='36' fill='%237c4dff'/></svg>"}" />
      <div>
        <div class="duet-name">${escHtml(u.displayName||u.email)}</div>
        <div class="duet-status">${(u.songsThisCycle||0) >= rules.songsPerCycle ? "Ja cantou neste ciclo" : "Disponivel"}</div>
      </div>
    </div>`).join("");
}

window.toggleDuet = function(uid, el) {
  const idx = selectedDuetIds.indexOf(uid);
  if (idx > -1) { selectedDuetIds.splice(idx,1); el.classList.remove("selected"); }
  else           { selectedDuetIds.push(uid);     el.classList.add("selected"); }
};

// -- Solicitar musica ----------------------------------------------------------
window.requestSong = async function() {
  const song   = document.getElementById("req-song").value.trim();
  const artist = document.getElementById("req-artist").value.trim();
  if (!song || !artist) { toast("Preencha o nome da musica e do artista"); return; }
  if (!currentUser) return;

  const myInQueue = queueData.some(q => q.userId === currentUser.uid);
  if (myInQueue) { toast("Voce ja esta na fila!"); return; }

  const songsUsed = userData?.songsThisCycle || 0;
  if (songsUsed >= rules.songsPerCycle) { toast("Voce atingiu o limite de musicas deste ciclo"); return; }

  if (queueData.length >= rules.maxQueue) { toast("A fila esta cheia. Aguarde."); return; }

  await addDoc(collection(db, "queue"), {
    song, artist,
    userId:      currentUser.uid,
    singerName:  userData?.displayName || currentUser.email,
    singerPhoto: userData?.photoURL || "",
    duetWith:    [],
    requestedAt: serverTimestamp(),
    status:      "waiting",
  });

  document.getElementById("req-song").value   = "";
  document.getElementById("req-artist").value = "";
  toast("Musica adicionada a fila! ?");
};

// -- Convidar para dueto -------------------------------------------------------
window.sendDuetInvite = async function() {
  if (!selectedDuetIds.length) { toast("Selecione pelo menos uma pessoa para o dueto"); return; }
  const myEntry = queueData.find(q => q.userId === currentUser.uid);
  if (!myEntry) { toast("Voce precisa ter uma musica na fila para convidar para dueto"); return; }

  const names = selectedDuetIds.map(id => {
    const snap = document.querySelector(`.duet-item.selected .duet-name`);
    return snap?.textContent || id;
  });

  await updateDoc(doc(db, "queue", myEntry.id), { duetWith: names });
  toast("Convite de dueto enviado! ?");
  selectedDuetIds = [];
  document.querySelectorAll(".duet-item.selected").forEach(el => el.classList.remove("selected"));
};

function escHtml(s) {
  if (!s) return "";
  return s.replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;").replace(/"/g,"&quot;");
}
